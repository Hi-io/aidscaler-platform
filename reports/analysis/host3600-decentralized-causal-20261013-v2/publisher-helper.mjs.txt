// Publication of an already closed, authorized three-world readout.
// No world/evaluator imports, provider calls, new effects or outcome selection.
import {readFile,writeFile,mkdir} from 'node:fs/promises';
import {createHash} from 'node:crypto';
import {resolve,relative} from 'node:path';
import {fileURLToPath} from 'node:url';

const sha=bytes=>createHash('sha256').update(bytes).digest('hex');
const previous='.local/reports/host3600-decentralized-causal-20261013-v1';
const destination='reports/analysis/host3600-decentralized-causal-20261013-v2';
const originalInput='reports/analysis/host3600-decentralized-closed-20261013.json';
const sourcePins=Object.freeze({candidate:'9b4de8b7769c69a3279067a8a093ef6d2392ec8a9728e704f871a184a69934e3',
 humanEvaluator:'75c635a59eda752820aabef317a42799915a46a5fc7fba82ad9d0382d69ea29e',
 simulator:'561e6a02fbde1d047f999c4f077cbff83c438ba5c508c4d8d18bbf59ff78e94b'});
const arms=['C_manual','C_decentralized','B6'];
const read=async path=>{const bytes=await readFile(path);return{path,bytes,sha256:sha(bytes)};};
const data=async path=>{const entry=await read(path);return{...entry,value:JSON.parse(entry.bytes.toString('utf8'))};};
const old=await data(previous+'/readout.json'),supplement=await data(previous+'/source-definitions-clarification.json');
const sealed=await data(originalInput),doc=await read(previous+'/readout.md');
const check=(condition,code)=>{if(!condition)throw Error(code);};
check(sealed.sha256===old.value.input.sha256,'PREDECESSOR_INPUT_CHANGED');
check(sealed.value.validRuns===3&&sealed.value.completedRuns===3&&sealed.value.rows?.length===3,'EXACT_CLOSED_GROUP_REQUIRED');
check(old.value.rows?.map(x=>x.arm).join(',')===arms.join(',')&&supplement.value.rows?.map(x=>x.arm).join(',')===arms.join(','),'ARM_INVENTORY_CHANGED');
check(old.value.source.stable&&old.value.source.reasons.length===0,'SOURCE_NOT_STABLE');
for(const epoch of ['initial','final'])check(old.value.source[epoch].candidate===sourcePins.candidate&&
 old.value.source[epoch].evaluator===sourcePins.humanEvaluator&&old.value.source[epoch].simulator===sourcePins.simulator,'SOURCE_PIN_CHANGED');
check(JSON.stringify(old.value.comparisons)===JSON.stringify(sealed.value.comparisons),'EFFECTS_CHANGED');

const rows=[];
for(const row of old.value.rows){
 check(row.status==='completed'&&row.file.replaceAll('\\','/').includes('/native-bridge-host3600-v2-20261013/decentralized/'),'OUTSIDE_CLOSED_AUTHORIZED_SCOPE');
 const raw=await data(row.file),extra=supplement.value.rows.find(x=>x.arm===row.arm);
 check(raw.sha256===row.sha256&&extra.rawSha256===row.sha256,'RAW_OR_SUPPLEMENT_CHANGED');
 check(raw.value.status==='completed'&&raw.value.protocol.wallLimitMs===3600000,'RAW_NOT_CLOSED_HOST3600');
 const renamedStages={};
 for(const [stage,item]of Object.entries(row.actualNativeStages)){
  const native=raw.value.metrics.humanCapacity.stages[stage],exact=extra.nativeStages[stage];
  check(item.completedQueueP95Ms===native.startedQueueMs.p95&&JSON.stringify(exact.startedQueueMs)===JSON.stringify(native.startedQueueMs),'STARTED_QUEUE_SOURCE_CHANGED');
  check(JSON.stringify(exact.completedTurnaroundMs)===JSON.stringify(native.finishedTurnaroundMs),'FINISHED_QUEUE_SOURCE_CHANGED');
  const {completedQueueP95Ms,...rest}=item;
  renamedStages[stage]={...rest,startedQueueMsIncludingWorkingExplicit:{...exact.startedQueueMs,
    definition:'Native wait at work start: samples include completed and still-working tasks; pending unstarted tasks are separately censored.',
    nativeField:'metrics.humanCapacity.stages.'+stage+'.startedQueueMs',workingAtCutoff:exact.working,
    completedAtCutoff:exact.completed,queuedAtCutoff:exact.queued,pendingCensored:exact.pendingCensored,cutoffMs:exact.cutoffMs},
   finishedP95Scope:{nativeField:'metrics.humanCapacity.stages.'+stage+'.finishedTurnaroundMs',samples:exact.completedTurnaroundMs.samples,
    definition:'Completed-task turnaround; pending tasks are not assigned an invented finishing time.'}};
 }
 rows.push({...row,actualNativeStages:renamedStages,issuedOperations:row.operationCount,
  operationCountDefinition:'All issued native operations, including orders still dispatched at cutoff; not completed receipts or deficit-reducing benefits.',
  operationStates:extra.operationStates,positivePhysicalReceipts:extra.positivePhysicalReceipts,
  positivePhysicalReceiptDefinition:'Received quantity greater than zero, not independently useful incremental benefit.'});
}

const helper=await read(fileURLToPath(import.meta.url)),oldHelper=await read('.local/analyze-host3600-decentralized-closed.mjs');
const companion=await read('.local/human-kpi-companion.mjs');
check(companion.sha256==='5f4c5e07ce21d52b2acaca0065f252409c43db80a317cffacecb7b41e85a5e2b','SEALED_COMPANION_CHANGED');
check(!JSON.stringify(rows).includes('"completedQueueP95Ms"'),'DEPRECATED_NAME_RETAINED_AS_MEASUREMENT');
const report={...old.value,schema:'closed-three-world-host3600-causal-publication-v2',rows,
 publication:{role:'Append-only presentation/definition correction after closed authorized read; original estimators and every effect are copied unchanged.',
  createdAt:new Date().toISOString(),correction:{previousField:'actualNativeStages.*.completedQueueP95Ms',
   exactField:'actualNativeStages.*.startedQueueMsIncludingWorkingExplicit',changesMeasurement:false,changesEstimator:false,
   changesEffect:false,excludesWorking:false,sourceSupplement:{path:supplement.path,sha256:supplement.sha256},
   explanation:'The v1 auxiliary name wrongly described native startedQueueMs as completed-only. The native samples include active work. No sealed reader/companion or raw result was edited.'},
  noNewSelection:true,worldsStarted:0,providerCalls:0,scope:'Only C_manual/C_decentralized/B6 closed decentralized worlds; no stress outcome access.',
  predecessor:{path:old.path,sha256:old.sha256},originalClosedReader:{path:sealed.path,sha256:sealed.sha256}}};

// Keeping predecessors explicitly marked avoids silently replacing a historical label.
await mkdir(destination,{recursive:false});await mkdir(destination+'/provenance',{recursive:false});
const files=[];
const put=async(path,bytes,role)=>{await writeFile(resolve(destination,path),bytes,{flag:'wx'});files.push({path,sha256:sha(bytes),bytes:bytes.length,role});};
await put('readout.json',Buffer.from(JSON.stringify(report,null,2)+'\n'),'Corrected publication; original effects copied unchanged');
const publicDoc=doc.bytes.toString('utf8').replace('# HOST3600: tres mundos cerrados del control descentralizado',
 '# HOST3600: diagnóstico causal de tres mundos cerrados — publicación v2').replace('y en los raws inmutables. El reader y companion sellados no se modificaron.',
 'y en los raws inmutables. En esta publicación el campo se llama `startedQueueMsIncludingWorkingExplicit` y explicita los denominadores; el alias auxiliar antiguo sólo se conserva como antecedente. El reader y companion sellados no se modificaron.');
await put('readout.md',Buffer.from(publicDoc),'Public causal discussion and limitations');
await put('source-definitions-clarification.json',supplement.bytes,'Exact native pending/start/finish samples and completed receipt counts; supplement preserved');
await put('publisher-helper.mjs.txt',helper.bytes,'Exact publisher bytes; copies effects and corrects labels without source imports');
await put('provenance/original-v1-helper.mjs.txt',oldHelper.bytes,'Historical helper only; deprecated name explicitly superseded, not current definition');
await put('provenance/original-v1-readout.json',old.bytes,'Unmodified predecessor, not current metric naming');
await put('provenance/original-v1-readout.md',doc.bytes,'Unmodified preceding explanation');
const metadata={schema:'append-only-closed-three-world-publication-files-v2',sourcePins,originalClosedReader:{path:sealed.path,sha256:sealed.sha256},
 rawInputs:rows.map(x=>({arm:x.arm,path:relative(process.cwd(),x.file).replaceAll('\\','/'),sha256:x.sha256,status:x.status})),
 antecedents:[{path:old.path,sha256:old.sha256},{path:supplement.path,sha256:supplement.sha256},{path:oldHelper.path,sha256:oldHelper.sha256}],
 sealedCompanion:{path:companion.path,sha256:companion.sha256},publisher:{path:helper.path,sha256:helper.sha256},files,
 verification:{originalEffectsCopiedExactly:true,measurementsReestimated:false,estimatorsChanged:false,resultsFiltered:false,
  rawsRehashed:true,newCoreWrites:0,newWorlds:0,newProviderCalls:0,stressOutcomesRead:false}};
await writeFile(resolve(destination,'manifest.json'),JSON.stringify(metadata,null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify({folder:resolve(destination),manifestSha256:sha(await readFile(resolve(destination,'manifest.json'))),
 publisherSha256:helper.sha256,readoutSha256:files.find(x=>x.path==='readout.json').sha256,worldsStarted:0,providerCalls:0}));
